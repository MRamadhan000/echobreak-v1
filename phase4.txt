            ctx.closePath();
        }

        function wrapText(ctx, text, x, y, maxWidth, lineHeight, maxLines) {
            const words = text.split(" ");
            let line = "", lines = 0;
            for (let i = 0; i < words.length; i++) {
                const test = line + words[i] + " ";
                if (ctx.measureText(test).width > maxWidth && line !== "") {
                    ctx.fillText(line.trim(), x, y);
                    line = words[i] + " ";
                    y += lineHeight;
                    lines++;
                    if (maxLines && lines >= maxLines - 1) {
                        ctx.fillText(line.trim() + "…", x, y);
                        return;
                    }
                } else {
                    line = test;
                }
            }
            ctx.fillText(line.trim(), x, y);
        }

        function hexToRgb(hex) {
            hex = hex.replace("#", "");
            const n = parseInt(hex, 16);
            return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
        }

        function makeChamberWallCanvas(accentHex) {
            const [r, g, b] = hexToRgb(accentHex);
            const size = 900;
            const c = document.createElement("canvas");
            c.width = size; c.height = size;
            const ctx = c.getContext("2d");
            const cx = size / 2, cy = size / 2, rad = size / 2 - 10;

            const grad = ctx.createRadialGradient(cx, cy, rad * 0.05, cx, cy, rad);
            grad.addColorStop(0, "rgba(255,255,255,0.95)");
            grad.addColorStop(0.25, `rgba(${r},${g},${b},0.55)`);
            grad.addColorStop(0.55, `rgba(${r},${g},${b},0.5)`);
            grad.addColorStop(0.8, `rgba(${Math.round(r*0.4)},${Math.round(g*0.4)},${Math.round(b*0.4)},0.32)`);

            ctx.beginPath();
            ctx.arc(cx, cy, rad, 0, Math.PI * 2);
            ctx.fillStyle = grad;
            ctx.fill();

            ctx.save();
            ctx.beginPath();
            ctx.arc(cx, cy, rad, 0, Math.PI * 2);
            ctx.clip();

            ctx.lineWidth = 2;
            for (let x = 0; x <= size; x += 60) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, size); ctx.stroke(); }
            for (let y = 0; y <= size; y += 60) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(size, y); ctx.stroke(); }
            ctx.restore();

            ctx.shadowColor = accentHex;
            ctx.shadowBlur = 60;
            ctx.beginPath();
            ctx.arc(cx, cy, rad - 4, 0, Math.PI * 2);
            ctx.lineWidth = 14;
            ctx.strokeStyle = accentHex;
            ctx.stroke();

            ctx.beginPath();
            ctx.arc(cx, cy, rad - 4, 0, Math.PI * 2);

        function getMesh(entity) {
            return entity.getObject3D("mesh");
        }

        function applyTexture(entity, canvas, opacity = 1) {
            const THREE = AFRAME.THREE;
            const setTex = () => {
                const mesh = getMesh(entity);
                if (!mesh) return;
                const tex = new THREE.CanvasTexture(canvas);
                tex.needsUpdate = true;
                mesh.material.map = tex;
                mesh.material.transparent = true;
                mesh.material.opacity = opacity;
                mesh.material.needsUpdate = true;
            };
            if (getMesh(entity)) setTex();
            else entity.addEventListener("loaded", setTex, { once: true });
        }

        function tweenOpacity(entity, from, to, duration, onDone) {
            const start = performance.now();
            function step(now) {
                const mesh = getMesh(entity);
                if (!mesh) return;
                const t = Math.min(1, (now - start) / duration);
                const eased = t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
                mesh.material.opacity = from + (to - from) * eased;
                mesh.material.transparent = true;
                mesh.material.needsUpdate = true;
                if (t < 1) requestAnimationFrame(step);
                else if (onDone) onDone();
            }
            requestAnimationFrame(step);
        }

        function tweenLineOpacity(entity, from, to, duration, onDone) {
            const start = performance.now();
            function step(now) {
                const t = Math.min(1, (now - start) / duration);
                const eased = t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2;
                const val = from + (to - from) * eased;
                entity.setAttribute("line", "opacity", val);
                if (t < 1) requestAnimationFrame(step);
                else if (onDone) onDone();
            }
            requestAnimationFrame(step);
        }

                    tweenLineOpacity(line, 0, 0.85, 420);

                    arrow.setAttribute("visible", "true");
                    tweenOpacity(arrow, 0, 1, 420);
                }, i * 220);
            });

            const totalDelay = REINFORCE_POSITIONS.length * 220 + 450;
            setTimeout(() => {
                appState = "revealed";
                refreshUnlockButtonState();
            }, totalDelay);
        }

        // ============================================================
        // KLIK UNLOCK KEDUA — urutan animasi unlock penuh:
        // 1) 4 kartu rekomendasi + garis + panah memudar & hancur, chamber
        //    (glow/wall/rim/scan) mengempis kembali & warna jadi hijau.
        // 2) Ikon gembok memantul & berputar, lalu berganti versi terbuka.
        // 3) 6 kartu komentar luar memudar (stagger), impact flash ikut mati.
        // 4) Berita utama crossfade ke data asli.
        // 5) State -> 'unlocked'.
        // ============================================================
        function runUnlockSequence() {
            if (appState !== "revealed" || !markerVisible) return;
            appState = "unlocking";
            refreshUnlockButtonState();

            lockStatus.innerText = "🔓 MEMBUKA...";
            lockStatus.style.color = "#facc15";
            setContext(
                "Fase 4: Membongkar Tembok Algoritma",
                "Sistem sedang menembus filter echo chamber dan menghapus komentar serta rekomendasi yang sebelumnya memperkuat gelembungmu...",
                "none"
            );

            bubbleGroup.setAttribute("animation__shrink",
                "property: scale; to: 1 1 1; dur: 500; easing: easeInOutQuad");
            updateBlockedCardBounce(1);
            updateChamberColors(ACCENT.unlocked);

            REINFORCE_POSITIONS.forEach((pos, i) => {
                const card = document.getElementById(pos.id);
                const line = document.getElementById(`${pos.id}-line`);
                const arrow = document.getElementById(`${pos.id}-arrow`);
                setTimeout(() => {
                    tweenOpacity(card, 1, 0, 400, () => card.setAttribute("visible", "false"));
                    tweenOpacity(arrow, 1, 0, 400, () => arrow.setAttribute("visible", "false"));
                    tweenLineOpacity(line, 0.85, 0, 400, () => line.setAttribute("visible", "false"));
                }, i * 130);
            });

            const lockIcon = document.getElementById("lockIcon");

            lockIcon.setAttribute("animation__pulse",
                "property: scale; from: 1 1 1; to: 1.35 1.35 1.35; dur: 350; dir: alternate; loop: 1; easing: easeOutBack");
            lockIcon.setAttribute("animation__spin",
                "property: rotation; from: 0 0 0; to: 0 0 25; dur: 350; dir: alternate; loop: 1; easing: easeInOutQuad");

            setTimeout(() => {
                applyTexture(lockIcon, makeLockCanvas(true));
                lockIcon.setAttribute("animation__open",
                    "property: scale; from: 1.1 1.1 1.1; to: 1.5 1.5 1.5; dur: 500; easing: easeOutQuad");

                setTimeout(() => {
                    tweenOpacity(lockIcon, 1, 0, 500, () => {
                        lockIcon.setAttribute("visible", "false");
                    });
                }, 550);
            }, 400);

            CARD_IDS.forEach((id, i) => {
                const el = document.getElementById(id);
                const impactEl = document.getElementById("impact-" + id.split("-")[1]);
                setTimeout(() => {
                    tweenOpacity(el, 1, 0, 550, () => {
                        el.setAttribute("visible", "false");
                        el.removeAttribute("animation");
                    });
                    if (impactEl) {
                        tweenOpacity(impactEl, impactEl.getObject3D("mesh") ? impactEl.getObject3D("mesh").material.opacity : 0.95, 0, 550, () => {
                            impactEl.setAttribute("visible", "false");
                            impactEl.removeAttribute("animation");
                            impactEl.removeAttribute("animation__scale");
                        });
                    }
                }, 300 + i * 160);
            });

            const bubble = document.getElementById("bubblePlane");
            const dataRevealDelay = 300 + CARD_IDS.length * 160 + 550 + 150;

            setTimeout(() => {
                tweenOpacity(bubble, 1, 0, 260, () => {
                    applyTexture(bubble, makeUnlockedBubbleCanvas(), 0);
                    tweenOpacity(bubble, 0, 1, 700, () => {
                        appState = "unlocked";
                        lockStatus.innerText = "🔓 TERBUKA";
                        lockStatus.style.color = "#4ade80";
                        debugPanel.classList.remove("reinforced");
                        debugPanel.classList.add("unlocked");
                        setContext(
                            "Fase 4: Data Asli Terungkap",
                            "Tembok algoritma berhasil dibongkar. Statistik dan fakta lengkap tentang jam masak & distribusi MBG kini ditampilkan.",
                            "unlocked"
                        );
                        refreshUnlockButtonState();
                    });
                });
            }, dataRevealDelay);
        }

        unlockButton.addEventListener("click", () => {
            if (appState === "locked") {
                runReinforceSequence();
            } else if (appState === "revealed") {
                runUnlockSequence();
            }
        });
    </script>
