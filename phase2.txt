        function roundRect(ctx, x, y, w, h, r) {
            ctx.beginPath();
            ctx.moveTo(x + r, y);
            ctx.arcTo(x + w, y, x + w, y + h, r);
            ctx.arcTo(x + w, y + h, x, y + h, r);
            ctx.arcTo(x, y + h, x, y, r);
            ctx.arcTo(x, y, x + w, y, r);
            ctx.closePath();
        }

        function wrapText(ctx, text, x, y, maxWidth, lineHeight, maxLines) {
            const words = text.split(" ");
            let line = "", lines = 0;
            for (let i = 0; i < words.length; i++) {
                const test = line + words[i] + " ";
                if (ctx.measureText(test).width > maxWidth && line !== "") {
                    ctx.fillText(line.trim(), x, y);
                    line = words[i] + " ";
                    y += lineHeight;
                    lines++;
                    if (maxLines && lines >= maxLines - 1) {
                        ctx.fillText(line.trim() + "…", x, y);
                        return;
                    }
                } else {
                    line = test;
                }
            }
            ctx.fillText(line.trim(), x, y);
        }

        // ---------- Dinding / Medan Gaya Algoritma (barrier ring) ----------
        function makeBarrierCanvas() {
            const size = 900;
            const c = document.createElement("canvas");
            c.width = size; c.height = size;
            const ctx = c.getContext("2d");
            const cx = size / 2, cy = size / 2;

            ctx.strokeStyle = "rgba(0, 212, 255, 0.55)";
            ctx.lineWidth = 3;
            ctx.setLineDash([26, 18]);
            ctx.beginPath();
            ctx.arc(cx, cy, size * 0.46, 0, Math.PI * 2);
            ctx.stroke();

            ctx.setLineDash([]);
            ctx.strokeStyle = "rgba(0, 212, 255, 0.22)";
            ctx.lineWidth = 1.5;
            ctx.beginPath();
            ctx.arc(cx, cy, size * 0.40, 0, Math.PI * 2);
            ctx.stroke();

            // tick marks like a radar / HUD
            const ticks = 48;
            for (let i = 0; i < ticks; i++) {
                const a = (i / ticks) * Math.PI * 2;
                const r1 = size * 0.46;
                const r2 = i % 4 === 0 ? size * 0.485 : size * 0.47;
                ctx.strokeStyle = i % 4 === 0 ? "rgba(0,212,255,0.6)" : "rgba(0,212,255,0.25)";
                ctx.lineWidth = i % 4 === 0 ? 2.5 : 1.2;
                ctx.beginPath();
                ctx.moveTo(cx + Math.cos(a) * r1, cy + Math.sin(a) * r1);
                ctx.lineTo(cx + Math.cos(a) * r2, cy + Math.sin(a) * r2);
                ctx.stroke();
            }

            return c;
        }

        // ---------- Percikan tolakan di tepi kartu (spark) ----------
        function makeSparkCanvas() {
            const size = 240;
            const c = document.createElement("canvas");
            c.width = size; c.height = size;
            const ctx = c.getContext("2d");
            const cx = size / 2, cy = size / 2;

            const grad = ctx.createRadialGradient(cx, cy, 2, cx, cy, size / 2);
            grad.addColorStop(0, "rgba(255,255,255,0.95)");
            grad.addColorStop(0.25, "rgba(255,45,85,0.85)");
            grad.addColorStop(1, "rgba(255,45,85,0)");
            ctx.fillStyle = grad;
            ctx.beginPath();
            ctx.arc(cx, cy, size / 2, 0, Math.PI * 2);
            ctx.fill();

            ctx.strokeStyle = "rgba(255,255,255,0.9)";
            ctx.lineWidth = 3;
            for (let i = 0; i < 6; i++) {
                const a = (i / 6) * Math.PI * 2;
                ctx.beginPath();
                ctx.moveTo(cx + Math.cos(a) * size * 0.10, cy + Math.sin(a) * size * 0.10);
                ctx.lineTo(cx + Math.cos(a) * size * 0.32, cy + Math.sin(a) * size * 0.32);
                ctx.stroke();
            }
            return c;
        }

        // ---------- Bubble Chat Utama (berita terkurung) ----------
        function makeBubbleCanvas() {
            const w = 680, h = 560;
            const c = document.createElement("canvas");
            c.width = w; c.height = h;
            const ctx = c.getContext("2d");

            // aura medan gaya di belakang bubble
            ctx.save();
            const aura = ctx.createRadialGradient(w/2, h/2 - 20, 40, w/2, h/2 - 20, w * 0.62);
            aura.addColorStop(0, "rgba(0,212,255,0.28)");
            aura.addColorStop(1, "rgba(0,212,255,0)");
            ctx.fillStyle = aura;
            ctx.fillRect(0, 0, w, h);
            ctx.restore();

            // shadow untuk bubble
            ctx.save();
            ctx.shadowColor = "rgba(0,212,255,0.55)";
            ctx.shadowBlur = 42;

        function applyTexture(entity, canvas) {
            const THREE = AFRAME.THREE;
            const setTex = () => {
                const mesh = entity.getObject3D("mesh");
                if (!mesh) return;
                const tex = new THREE.CanvasTexture(canvas);
                tex.needsUpdate = true;
                mesh.material.map = tex;
                mesh.material.transparent = true;
                mesh.material.needsUpdate = true;
            };
            if (entity.getObject3D("mesh")) setTex();
            else entity.addEventListener("loaded", setTex, { once: true });
        }

        async function buildTextures() {
            if (document.fonts && document.fonts.ready) await document.fonts.ready;

            applyTexture(document.getElementById("barrierRing"), makeBarrierCanvas());
            applyTexture(document.getElementById("bubblePlane"), makeBubbleCanvas());

            const sparkTex = makeSparkCanvas();
            Object.keys(COMMENTS).forEach((id) => {
                applyTexture(document.getElementById(id), makeCardCanvas(COMMENTS[id]));
                const sparkId = "spark-" + id.split("-")[1];
                const sparkEl = document.getElementById(sparkId);
                if (sparkEl) applyTexture(sparkEl, sparkTex);
            });
        }
        buildTextures();
